const AUCTION_SETUP_SHEET = "Auction_Setup"; const BIDS_SHEET = "Bids"; const BLOCKED_SHEET = "Blocked"; function doGet() { const auction = getCurrentAuction(); const bids = getValidBidsForAuction(auction.id); return responseData({ auction: auction, bids: bids, highestBid: bids.length ? bids[0].bidAmount : auction.startingBid }); } function doPost(e) { const auction = getCurrentAuction(); if (auction.status !== "OPEN") { return response(false, "Auction is not open."); } const now = new Date(); const closing = new Date(auction.closingTime); if (now > closing) { return response(false, "Auction closed. No further bids accepted."); } let data; try { data = JSON.parse(e.postData.contents); } catch (err) { return response(false, "Invalid request format."); } const firstName = cleanText(data.firstName); const surnameInitial = cleanText(data.surnameInitial).toUpperCase(); const phone = cleanPhone(data.phone); const transferMethod = cleanText(data.transferMethod); const paymentTiming = cleanText(data.paymentTiming); const bidAmount = Number(data.bidAmount); const acceptedRules = data.acceptedRules === true; if (!acceptedRules) { return response(false, "You must accept the auction rules before bidding."); } if (!isValidName(firstName)) { return response(false, "Please enter a valid first name."); } if (!isValidInitial(surnameInitial)) { return response(false, "Please enter one surname initial only."); } if (!isValidJamaicaPhone(phone)) { return response(false, "Please enter a valid Jamaica phone number starting with 876 or 658."); } if (isBlocked(phone)) { return response(false, "This phone number is blocked from bidding."); } if (transferMethod.length < 2) { return response(false, "Please enter the bank or transfer agency you will use."); } if (paymentTiming.length < 5) { return response(false, "Please state when you can complete payment if you win."); } if (!bidAmount || bidAmount < auction.startingBid) { return response(false, "Bid is below the starting bid."); } const bids = getValidBidsForAuction(auction.id); const minimumRequiredBid = bids.length ? bids[0].bidAmount + auction.minimumIncrease : auction.startingBid; if (bidAmount < minimumRequiredBid) { return response( false, "Bid too low. Your bid must be at least J$" + minimumRequiredBid.toLocaleString() + "." ); } const identityIssue = getPriorBidIdentityIssue( auction.id, phone, firstName, surnameInitial, transferMethod ); if (identityIssue && data.confirmIdentityChange !== true) { return response( false, "This phone number has already been used for this auction with different bidder details.\n\nPlease check that your name, surname initial, and bank selection are correct before bidding again.", { requiresIdentityConfirmation: true, identityIssue: identityIssue } ); } const bidderId = createBidderId(auction.id, phone); const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(BIDS_SHEET); const identityWarningFlag = identityIssue && data.confirmIdentityChange === true ? identityIssue.warningFlag : "NORMAL"; sheet.appendRow([ new Date(), auction.id, auction.productName, firstName, surnameInitial, phone, transferMethod, paymentTiming, bidAmount, bidderId, "VALID", identityWarningFlag ]); return response(true, "Bid accepted.", { bidderId: bidderId, bidAmount: bidAmount }); } function getCurrentAuction() { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(AUCTION_SETUP_SHEET); const row = sheet.getRange(2, 1, 1, 15).getValues()[0]; return { id: cleanText(row[0]), productName: cleanText(row[1]), shortDescription: cleanText(row[2]), fullDescription: cleanText(row[1]), productPhotoLink: cleanText(row[3]), bannerImageLink: cleanText(row[4]), introSoundLink: convertGoogleDriveLink(cleanText(row[5])), introMessage: cleanText(row[6]), startingBid: Number(row[7]), minimumIncrease: Number(row[8]), closingTime: row[9], paymentDeadlineMinutes: Number(row[10]), deliveryTerms: cleanText(row[11]), status: cleanText(row[12]).toUpperCase() }; } function getValidBidsForAuction(productId) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(BIDS_SHEET); const rows = sheet.getDataRange().getValues(); rows.shift(); let bids = []; rows.forEach(row => { if (row[1] === productId && row[10] === "VALID") { bids.push({ timestamp: row[0], bidderId: row[9], bidAmount: Number(row[8]) }); } }); bids.sort((a, b) => b.bidAmount - a.bidAmount); return bids; } function isBlocked(phone) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(BLOCKED_SHEET); const rows = sheet.getDataRange().getValues(); rows.shift(); return rows.some(row => cleanPhone(row[0]) === phone); } function createBidderId(productId, phone) { const raw = productId + "-" + phone; const digest = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, raw); const code = Math.abs(digest[0] * 256 + digest[1]).toString().slice(0, 3); return "Bidder " + code; } function cleanText(value) { return String(value || "").trim(); } function cleanPhone(value) { return String(value || "").replace(/\D/g, ""); } function isValidName(name) { return /^[A-Za-z]{2,20}$/.test(name); } function isValidInitial(initial) { return /^[A-Z]$/.test(initial); } function isValidJamaicaPhone(phone) { return /^(876|658)\d{7}$/.test(phone); } function convertGoogleDriveLink(link) { if (!link) return ""; const match = link.match(/\/d\/([^/]+)/); if (match && match[1]) { return "https://drive.google.com/thumbnail?id=" + match[1] + "&sz=w1000"; } return link; } function getPriorBidIdentityIssue(productId, phone, firstName, surnameInitial, transferMethod) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(BIDS_SHEET); const rows = sheet.getDataRange().getValues(); rows.shift(); const newFirstName = cleanText(firstName); const newInitial = cleanText(surnameInitial).toUpperCase(); const newTransferMethod = cleanText(transferMethod); for (let i = rows.length - 1; i >= 0; i--) { const row = rows[i]; if (row[1] === productId && cleanPhone(row[5]) === phone && row[10] === "VALID") { const previousFirstName = cleanText(row[3]); const previousInitial = cleanText(row[4]).toUpperCase(); const previousTransferMethod = cleanText(row[6]); const firstNameChanged = previousFirstName.toLowerCase() !== newFirstName.toLowerCase(); const initialChanged = previousInitial !== newInitial; const bankChanged = previousTransferMethod.toLowerCase() !== newTransferMethod.toLowerCase(); if (firstNameChanged || initialChanged || bankChanged) { return { firstNameChanged: firstNameChanged, initialChanged: initialChanged, bankChanged: bankChanged, previousFirstName: previousFirstName, previousInitial: previousInitial, previousTransferMethod: previousTransferMethod, newFirstName: newFirstName, newInitial: newInitial, newTransferMethod: newTransferMethod, warningFlag: firstNameChanged && initialChanged && bankChanged ? "FIRST NAME, INITIAL AND BANK CHANGED" : [ firstNameChanged ? "FIRST NAME CHANGED" : "", initialChanged ? "INITIAL CHANGED" : "", bankChanged ? "BANK CHANGED" : "" ].filter(Boolean).join(" + ") }; } return null; } } return null; } function response(success, message, extra) { const output = Object.assign( { success: success, message: message }, extra || {} ); return ContentService .createTextOutput(JSON.stringify(output)) .setMimeType(ContentService.MimeType.JSON); } function responseData(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
